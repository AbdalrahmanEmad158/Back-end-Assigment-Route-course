db.books.aggregate([
  {
    $addFields: {
      bookIdString: { $toString: "$_id" }
    }
  },
  {
    $lookup: {
      from: "logs",
      localField: "bookIdString",
      foreignField: "book_id",
      as: "logs"
    }
  },
  {
    $unwind: "$logs"
  },
  {
    $group: {
      _id: "$logs.action",
      book_details: {
        $push: {
          title: "$title",
          author: "$author",
          year: "$year"
        }
      }
    }
  },
  {
    $project: {
      _id: 0,
      action: "$_id",
      book_details: 1
    }
  }
])

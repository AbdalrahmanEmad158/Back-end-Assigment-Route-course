    var test = init
    function reset(){
        test = init
            console.log(test)
            return test
        }


    function increment(){
        test= test+1
        console.log(test)
        return test
    }


    function  decrement(){
        test-=1
            console.log(test)
            return test
        }

    return { reset: reset, increment: increment, decrement: decrement };
};

function findKthPositive(arr, k) {
    let current = 1;
    let index = 0;

    while (k > 0) {

        
        if (index < arr.length && arr[index] === current) {
            index++;
        } 
        
        
        
        
        else {
            k--;
            if (k === 0) {
                return current;
            }
        }

        current++;
    }
}

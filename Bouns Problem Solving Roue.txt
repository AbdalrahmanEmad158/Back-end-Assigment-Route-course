function findKthPositive(arr, k) {
    let current = 1;
    let index = 0;

    while (k > 0) {

        
        if (index < arr.length && arr[index] === current) {
            index++;
        } 
        
        
        
        
        else {
            k--;
            if (k === 0) {
                return current;
            }
        }

        current++;
    }

}

SELECT
    v.customer_id,
    COUNT(*) AS count_no_trans
FROM Visits v
LEFT JOIN Transactions t
    ON v.visit_id = t.visit_id
WHERE t.transaction_id IS NULL
GROUP BY v.customer_id;
